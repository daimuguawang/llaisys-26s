### 2.3 Runtime API核心实现

**文件：`src/device/nvidia/nvidia_runtime_api.cu`**

需要实现的API列表（对照`src/device/cpu/cpu_runtime_api.cpp`）：

```cpp
#include "src/device/runtime_api.hpp"
#include <cuda_runtime.h>
#include <cublas_v2.h>

namespace nvidia {

// ========== 内存管理 API ==========

static void* device_malloc(size_t size) {
    void* ptr = nullptr;
    cudaMalloc(&ptr, size);
    return ptr;
}

static void device_free(void* ptr) {
    cudaFree(ptr);
}

static void* host_malloc(size_t size) {
    void* ptr = nullptr;
    cudaMallocHost(&ptr, size);  // Pinned memory for async transfers
    return ptr;
}

static void host_free(void* ptr) {
    cudaFreeHost(ptr);
}

// ========== 内存拷贝 API ==========

static void memcpy_h2d(void* dst, const void* src, size_t size) {
    cudaMemcpy(dst, src, size, cudaMemcpyHostToDevice);
}

static void memcpy_d2h(void* dst, const void* src, size_t size) {
    cudaMemcpy(dst, src, size, cudaMemcpyDeviceToHost);
}

static void memcpy_d2d(void* dst, const void* src, size_t size) {
    cudaMemcpy(dst, src, size, cudaMemcpyDeviceToDevice);
}

// ========== 设备管理 API ==========

static void set_device(int device_id) {
    cudaSetDevice(device_id);
}

static int get_device() {
    int dev = 0;
    cudaGetDevice(&dev);
    return dev;
}

static int get_device_count() {
    int count = 0;
    cudaGetDeviceCount(&count);
    return count;
}

static void synchronize() {
    cudaDeviceSynchronize();
}

// ========== 资源管理 ==========

struct NvidiaResource {
    cublasHandle_t cublas_handle;
    cudaStream_t stream;
};

static NvidiaResource* create_resource() {
    auto* res = new NvidiaResource();
    cublasCreate(&res->cublas_handle);
    cudaStreamCreate(&res->stream);
    cublasSetStream(res->cublas_handle, res->stream);
    return res;
}

static void destroy_resource(void* resource) {
    auto* res = static_cast<NvidiaResource*>(resource);
    cublasDestroy(res->cublas_handle);
    cudaStreamDestroy(res->stream);
    delete res;
}

// ========== API 表注册 ==========

static LlaisysRuntimeAPI cuda_api = {
    .device_malloc = device_malloc,
    .device_free = device_free,
    .host_malloc = host_malloc,
    .host_free = host_free,
    .memcpy_h2d = memcpy_h2d,
    .memcpy_d2h = memcpy_d2h,
    .memcpy_d2d = memcpy_d2d,
    .set_device = set_device,
    .get_device = get_device,
    .get_device_count = get_device_count,
    .synchronize = synchronize,
    .create_resource = create_resource,
    .destroy_resource = destroy_resource,
};

const LlaisysRuntimeAPI* getRuntimeAPI() {
    return &cuda_api;
}

}  // namespace nvidia
```

### 3.1 通用算子模板

**文件：`src/ops/add/nvidia/add.cu`（参考实现）**

```cpp
#include "src/ops/add/op.hpp"
#include <cuda_runtime.h>

namespace llaisys {

__global__ void add_kernel(float* out, const float* a, const float* b, size_t n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        out[idx] = a[idx] + b[idx];
    }
}

// FP16版本
__global__ void add_kernel_half(half* out, const half* a, const half* b, size_t n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        out[idx] = __hadd(a[idx], b[idx]);
    }
}

void add_nvidia(tensor_t out, tensor_t a, tensor_t b) {
    size_t n = out->numel();
    int threads = 256;
    int blocks = (n + threads - 1) / threads;
    
    auto dtype = out->dtype();
    if (dtype == DataType::kFloat32) {
        add_kernel<<<blocks, threads>>>(
            out->data<float>(), a->data<float>(), b->data<float>(), n);
    } else if (dtype == DataType::kFloat16) {
        add_kernel_half<<<blocks, threads>>>(
            out->data<half>(), a->data<half>(), b->data<half>(), n);
    }
    // BF16类似...
}

}  // namespace llaisys
```

### 3.2 各算子实现要点

#### 3.2.1 Argmax
```cuda
// 并行规约求最大值
__global__ void argmax_kernel(const float* vals, int64_t* max_idx, float* max_val, int n) {
    __shared__ float s_max[256];
    __shared__ int s_idx[256];
    
    int tid = threadIdx.x;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    float local_max = -INFINITY;
    int local_idx = -1;
    
    while (idx < n) {
        if (vals[idx] > local_max) {
            local_max = vals[idx];
            local_idx = idx;
        }
        idx += blockDim.x * gridDim.x;
    }
    
    s_max[tid] = local_max;
    s_idx[tid] = local_idx;
    __syncthreads();
    
    // 规约
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) {
            if (s_max[tid + s] > s_max[tid]) {
                s_max[tid] = s_max[tid + s];
                s_idx[tid] = s_idx[tid + s];
            }
        }
        __syncthreads();
    }
    
    if (tid == 0) {
        *max_val = s_max[0];
        *max_idx = s_idx[0];
    }
}
```

#### 3.2.3 Linear（使用cuBLAS，性能关键）
```cpp
void linear_nvidia(tensor_t out, tensor_t in, tensor_t weight, tensor_t bias) {
    auto* res = static_cast<NvidiaResource*>(get_current_resource());
    cublasHandle_t handle = res->cublas_handle;
    
    // M = batch, N = out_features, K = in_features
    // out = in * weight^T + bias
    // cuBLAS使用列主序，所以计算 C = B * A^T
    
    int M = in->shape()[0];
    int K = in->shape()[1];
    int N = weight->shape()[0];
    
    float alpha = 1.0f, beta = 0.0f;
    
    // 使用cublasGemmEx支持混合精度
    cublasGemmEx(handle,
        CUBLAS_OP_T, CUBLAS_OP_N,
        N, M, K,
        &alpha,
        weight->data(), CUDA_R_16F, K,
        in->data(), CUDA_R_16F, K,
        &beta,
        out->data(), CUDA_R_16F, N,
        CUBLAS_COMPUTE_32F,
        CUBLAS_GEMM_DEFAULT_TENSOR_OP);
    
    // 添加bias
    if (bias) {
        // bias broadcast kernel
    }
}
```

#### 3.2.4 RMS Norm
```cuda
__global__ void rms_norm_kernel(
    float* out, const float* in, const float* weight,
    int rows, int cols, float eps) {
    int row = blockIdx.x;
    
    // 每行一个block，先求平方和
    __shared__ float s_sum;
    s_sum = 0.0f;
    __syncthreads();
    
    float sum = 0.0f;
    for (int j = threadIdx.x; j < cols; j += blockDim.x) {
        float x = in[row * cols + j];
        sum += x * x;
    }
    
    // warp reduce
    for (int offset = warpSize / 2; offset > 0; offset >>= 1)
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    
    if (threadIdx.x % warpSize == 0) atomicAdd(&s_sum, sum);
    __syncthreads();
    
    float rms = rsqrtf(s_sum / cols + eps);
    
    for (int j = threadIdx.x; j < cols; j += blockDim.x) {
        out[row * cols + j] = in[row * cols + j] * rms * weight[j];
    }
}
```

#### 3.2.5 RoPE
```cuda
__global__ void rope_kernel(
    float* out, const float* in, const int64_t* pos_ids,
    int seq_len, int n_heads, int d, float theta) {
    int s = blockIdx.x;
    int h = blockIdx.y;
    int j = threadIdx.x;
    
    if (j < d / 2) {
        int64_t pos = pos_ids[s];
        float freq = 1.0f / powf(theta, (2.0f * j) / d);
        float angle = pos * freq;
        float cos_val = cosf(angle);
        float sin_val = sinf(angle);
        
        int base_idx = (s * n_heads + h) * d;
        float a = in[base_idx + j];
        float b = in[base_idx + j + d/2];
        
        out[base_idx + j] = a * cos_val - b * sin_val;
        out[base_idx + j + d/2] = b * cos_val + a * sin_val;
    }
}
```

## 五、RTX 5090专项优化

### 5.1 Blackwell架构利用

1. **FP8张量核心**：
   - 线性层和注意力可使用FP8加速
   - 使用`cublasGemmEx` with `CUDA_R_8F_E4M3`

2. **线程簇（Thread Block Clusters）**：
   - 大矩阵乘法时使用分布式shared memory

3. **异步拷贝**：
   - 使用`cp.async`进行全局内存到共享内存的异步拷贝

### 5.2 显存优化（32GB配置）

32GB显存非常充裕，无需激进优化：
- KV Cache使用FP16，最大支持seq_len=32768
- 可开启Paged Attention减少碎片
- 权重可考虑FP8量化进一步提速

## 七、常见错误与解决方案

| 错误现象 | 原因 | 解决方案 |
|---------|------|---------|
| `no kernel image is available` | 算力架构不匹配 | 确认`add_cugencodes("sm_120")` |
| 算子结果与CPU不一致 | 线程同步问题 | kernel后添加`cudaDeviceSynchronize()` |
| cuBLAS结果错误 | 行列主序/转置问题 | 仔细核对A/B/C的leading dimension |
| 显存泄漏 | cudaMalloc未释放 | 使用RAII包装或确保device_free被调用 |
| 编译找不到cuda.h | CUDA路径未设置 | `export CUDA_HOME=/usr/local/cuda` |
